(() => {
  const out = {};
  const wt = document.getElementById('workTab');
  wt.click(); // navigate -> adds 'on'
  const chev = wt.querySelector('.tw-chev');
  out.wt_class = wt.className;
  out.chev_tag = chev.tagName;
  out.closest_group_on = !!chev.closest('.tab.group.on');
  out.closest_group = !!chev.closest('.tab.group');
  out.computed_display = getComputedStyle(chev).display;
  // inspect matched rules
  const matched = [];
  for (const sheet of document.styleSheets) {
    let rules;
    try { rules = sheet.cssRules; } catch(e){ continue; }
    if (!rules) continue;
    for (const r of rules) {
      if (r.media) { // recurse into media
        for (const mr of r.cssRules) {
          if (mr.selectorText && chev.matches(mr.selectorText) && /display/.test(mr.cssText)) matched.push('@media('+r.media.mediaText+') '+mr.selectorText+' => '+mr.style.display);
        }
        continue;
      }
      if (r.selectorText && chev.matches(r.selectorText) && /display/.test(r.cssText)) matched.push(r.selectorText+' => '+r.style.display);
    }
  }
  out.matched_display_rules = matched;
  return JSON.stringify(out, null, 2);
})()
