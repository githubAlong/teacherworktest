(() => {
  const out = {};
  const wt = document.getElementById('workTab');
  wt.click();
  const c = wt.querySelector('.tw-chev');
  if (c) c.dispatchEvent(new MouseEvent('click',{bubbles:true,cancelable:true}));
  out.wt_class = wt.className;
  out.closest_group_collapsed = !!c.closest('.tab.group.collapsed');
  out.computed_transform = c ? getComputedStyle(c).transform : 'n/a';
  const matched = [];
  for (const sheet of document.styleSheets) {
    let rules; try { rules = sheet.cssRules; } catch(e){ continue; }
    const walk = (list, media) => {
      for (const r of list) {
        if (r.media) { walk(r.cssRules, (media?media+' ':'')+r.media.mediaText); continue; }
        if (r.selectorText && c.matches(r.selectorText)) {
          if (/transform/.test(r.cssText) || r.style.transform) matched.push((media?'@media('+media+') ':'')+r.selectorText+' => transform:'+(r.style.transform||'(unset)'));
        }
      }
    };
    walk(rules, '');
  }
  out.matched_transform_rules = matched;
  return JSON.stringify(out, null, 2);
})()
